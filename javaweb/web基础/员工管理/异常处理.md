## 2.全局异常处理器

<img src="C:\Users\24709\AppData\Roaming\Typora\typora-user-images\image-20250619110549628.png" alt="image-20250619110549628" style="zoom:50%;" />

```java
/**
 * 全局异常处理
 * 异常获取是从下往上获取的
 */
@Slf4j
@RestControllerAdvice //声明当前是一个全局异常处理器
public class GlobalExcetionHandler {

    @ExceptionHandler // 声明当前方法可以处理哪些异常
    public Result handleException(Exception e) {
        log.error("服务器异常:",e);
        return Result.error("服务器异常，请联系管理员");
    }

    @ExceptionHandler // 声明当前方法可以处理DuplicateKeyException类型的异常
    public Result handleDuplicateKeyException(DuplicateKeyException e) {
        log.error("服务器异常:",e);
        String message = e.getMessage();// 获取异常信息
        int i = message.indexOf("Duplicate entry"); // 获取异常信息中包含的字符串所在位置
        String errorMsg = message.substring(i); // 通过位置截取异常信息
        String[] arr = errorMsg.split(" ");//将截取到的数组按照空格进行分割
        return Result.error("数据已存在：" + arr[2]);
    }
}
```

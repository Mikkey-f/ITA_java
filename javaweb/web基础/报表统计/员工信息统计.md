```java
@Service
public class ReportServiceImpl implements ReportService {
    @Autowired
    private EmpMapper empMapper;
    public JobOption getEmpJobData() {
        //1.调用mapper接口, 获取统计数据
        List<Map<String, Object>> list = empMapper.countEmpJobData(); // [{pos=教研主管, num=3}, {pos=讲师, data=2}, {pos=python, data=1}]

        //2.组装结果，并返回

//        List<String> jobList = new ArrayList<>();
//        List<Integer> dataList = new ArrayList<>();
//        for (Map<String, Object> map : list) {
//            jobList.add(map.get("pos").toString());
//            dataList.add(Integer.parseInt(map.get("num").toString()));
//        }
        
        //使用Java8的流式操作，将list中的每个map对象转换为两个独立的列表，分别存储职位名称和人数。
        //将数据库查询结果（list）中的职位名称（pos）和人数（num）分离为两个独立的列表，用于后续数据处理或前端展示。
        List<Object> jobList = list.stream().map(datamap -> datamap.get("pos")).toList();
        List<Object> dataList = list.stream().map(datamap -> datamap.get("num")).toList();
        return new JobOption(jobList, dataList);
    }
}
```



## 二.性别统计

![image-20250619171251386](C:\Users\24709\AppData\Roaming\Typora\typora-user-images\image-20250619171251386.png)

```mysql
-- 统计员工性别人数
-- if(条件,true_value,false_value)
select
    if(gender = 1,'男性员工','女性员工') name,
    count(*) value
from emp group by gender
```
